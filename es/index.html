01

Guarda la búsqueda en la URL

En vez de leer ActivatedRoute y llamar al router a mano, linkedQueryParam te da un signal que escribe en la URL, y una URL que escribe en el signal.

líneas 16 16 → 3
Leer la documentación de linkedQueryParam → 
export class SearchPage {
  private route = inject(ActivatedRoute);
  private router = inject(Router);

  search = toSignal(
    this.route.queryParamMap.pipe(
      map((params) => params.get('search')),
    ),
  );

  setSearch(search: string) {
    this.router.navigate([], {
      relativeTo: this.route,
      queryParams: { search },
      queryParamsHandling: 'merge',
    });
  }
}
export class SearchPage {
  search = linkedQueryParam('search');
}

constructor() {
  const controller = new AbortController();
  inject(DestroyRef).onDestroy(() =>
    controller.abort(),
  );

  window.addEventListener(
    'resize',
    (event) => {
      console.log('Window resized', event);
    },
    { signal: controller.signal },
  );
}

03

Guarda un ajuste en localStorage

injectLocalStorage lee el valor guardado, escribe cada cambio y sigue a las otras pestañas con el evento storage. Lo que recibes es un WritableSignal normal.

líneas 22 22 → 7
Leer la documentación de injectLocalStorage → 
export class ThemeToggle {
  readonly theme = signal<'light' | 'dark'>(
    JSON.parse(
      localStorage.getItem('theme') ?? '"light"',
    ),
  );

  constructor() {
    effect(() => {
      localStorage.setItem(
        'theme',
        JSON.stringify(this.theme()),
      );
    });

    fromEvent<StorageEvent>(window, 'storage')
      .pipe(takeUntilDestroyed())
      .subscribe((e) => {
        if (e.key === 'theme' && e.newValue) {
          this.theme.set(JSON.parse(e.newValue));
        }
      });
  }
}
export class ThemeToggle {
  readonly theme = injectLocalStorage<
    'light' | 'dark'
  >('theme', {
    defaultValue: 'light',
  });
}
